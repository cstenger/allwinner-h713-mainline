import os, re, time
from pathlib import Path

root=Path('/home/chris/Projects/h713/docs/hdmi-evidence/2026-10-17-uboot-output-diagnostic')
out=root/'preinit-validation'
log=root/'serial.log'
offset=log.stat().st_size
fd=os.open('/dev/ttyUSB0',os.O_WRONLY|os.O_NOCTTY)

def run(cmd,end,name,timeout=15):
    if len(cmd)>=500:raise RuntimeError('Console line too long')
    start=log.stat().st_size
    os.write(fd,(cmd+'\n').encode())
    deadline=time.monotonic()+timeout
    while time.monotonic()<deadline:
        text=log.read_bytes()[start:].replace(b'\r',b'').decode(errors='replace')
        if re.search(r'(?:\n|^)'+re.escape(end)+r'\n',text):
            (out/name).write_text(text)
            if any(s in text for s in ('Unknown command','Usage:','Error:')):raise RuntimeError('Console error '+name)
            return text
        time.sleep(.05)
    raise RuntimeError('No completed marker '+end)

try:
    print('Armed for next warm U-Boot prompt; NO cold-boot interception intended',flush=True)
    deadline=time.monotonic()+120
    sent=False
    while time.monotonic()<deadline:
        text=log.read_bytes()[offset:]
        if b'U-Boot 20' in text:
            if b'=> ' in text[text.find(b'U-Boot 20'):]:break
            if b'Hit any key' in text and not sent:
                os.write(fd,b' ');sent=True
        time.sleep(.05)
    else:raise RuntimeError('No U-Boot prompt within120s')
    state=run('\x15echo H713_PREINIT_CHECK_BEGIN; md.l 0x07090118 2; md.l 0x0306101c 1; echo H713_PREINIT_CHECK_END','H713_PREINIT_CHECK_END','state-before.txt')
    if '07090118: 48444d49 00000000' not in state or '0306101c: 00000000' not in state:
        raise RuntimeError('Unexpected warm state; left at prompt')
    print('Warm state verified; prepare/teardown before logo',flush=True)
    off=run('echo H713_PREINIT_OFF_BEGIN; gpio set PB5; sleep 5; h713_disp teardown; md.l 0x02000100 1; md.l 0x02000160 1; md.l 0x051c0014 1; md.l 0x051c0028 1; md.l 0x05140054 1; md.l 0x0200160c 1; echo H713_PREINIT_OFF_END','H713_PREINIT_OFF_END','controlled-off.txt',20)
    for expected in ('02000100: 00000000','02000160: 00000000','051c0014: 18000000','051c0028: 00000030','05140054: 40000000','0200160c: 00000000'):
        if expected not in off:raise RuntimeError('Off-state validation failed; left at prompt')
    print('Output at rest, reset asserted, PF6 low; holding3s',flush=True)
    held=run('sleep 3; echo H713_PREINIT_HOLD_DONE; md.l 0x02000100 1; md.l 0x02000160 1; echo H713_PREINIT_HOLD_END','H713_PREINIT_HOLD_END','off-hold.txt')
    if '02000100: 00000000' not in held or '02000160: 00000000' not in held:
        raise RuntimeError('Off hold did not persist; left at prompt')
    result=run('echo H713_PREINIT_NORMAL_LOGO_BEGIN; gpio clear PB5; if h713_disp auto 0x34 logo; then gpio set PB5; echo H713_PREINIT_LOGO_OK; else gpio set PB5; echo H713_PREINIT_LOGO_FAILED; fi; echo H713_PREINIT_LOGO_END','H713_PREINIT_LOGO_END','normal-logo.txt',20)
    if '\nH713_PREINIT_LOGO_OK\n' not in result:raise RuntimeError('Logo init failed; left at prompt')
    print('Normal logo initialization completed after explicit off sequence; Linux NOT started',flush=True)
finally:
    os.close(fd)
